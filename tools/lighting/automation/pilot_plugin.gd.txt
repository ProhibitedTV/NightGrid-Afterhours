@tool
extends EditorPlugin
## Isolated editor stage only; uses the native Godot 4.7.2 toolbar signal.
var started := false

func _enter_tree() -> void:
	call_deferred("_run")

func _find(node: Node, type_name: String, button_text: String = "") -> Array[Node]:
	var result: Array[Node] = []
	if node.is_class(type_name):
		if button_text.is_empty() or (node is Button and node.text == button_text and node.is_visible_in_tree()):
			result.append(node)
	for child in node.get_children(true):
		result.append_array(_find(child, type_name, button_text))
	return result

func _run() -> void:
	if started:
		return
	started = true
	if not FileAccess.file_exists("res://nightgrid_bake_pilot.marker"):
		_stop("Not a fresh isolated bake project", 1)
		return
	var inputs: Variant = JSON.parse_string(FileAccess.get_file_as_string("res://inputs.json"))
	if not inputs is Dictionary:
		_stop("Missing stage metadata", 1)
		return
	var relative_scene := str(inputs.get("scene"))
	if relative_scene != "tools/lighting/synaxis_reception_bake.tscn" and relative_scene != "scenes/synaxis/ground_envelope.tscn":
		_stop("Unknown bake scene", 1)
		return
	var scene_path := "res://" + relative_scene
	var output_path := scene_path.get_basename() + ".lmbake"
	if FileAccess.file_exists(output_path):
		_stop("Bake would overwrite an existing atlas", 1)
		return
	if not await _wait_filesystem_idle():
		_stop("Initial native import did not settle", 1)
		return
	EditorInterface.open_scene_from_path(scene_path)
	for i in range(300):
		await get_tree().process_frame
		var scene := EditorInterface.get_edited_scene_root()
		if scene != null and scene.scene_file_path == scene_path:
			break
	var scene := EditorInterface.get_edited_scene_root()
	if scene == null or scene.scene_file_path != scene_path:
		_stop("Pilot scene did not open", 1)
		return
	var gi := scene.get_node_or_null(str(inputs.get("gi_node"))) as LightmapGI
	if gi == null or gi.light_data != null:
		_stop("Expected unbaked LightmapGI", 1)
		return
	gi.set_bake_quality(LightmapGI.BAKE_QUALITY_HIGH)
	print("NIGHTGRID_BAKE_PILOT: High quality selected")
	EditorInterface.get_selection().clear()
	EditorInterface.get_selection().add_node(gi)
	EditorInterface.edit_node(gi)
	EditorInterface.set_main_screen_editor("3D")
	var buttons: Array[Node] = []
	for i in range(300):
		await get_tree().process_frame
		buttons = _find(EditorInterface.get_base_control(), "Button", "Bake Lightmaps")
		if buttons.size() == 1:
			break
	if buttons.size() != 1 or (buttons[0] as Button).disabled:
		_stop("Native Bake Lightmaps toolbar unavailable or ambiguous", 1)
		return
	var button := buttons[0] as Button
	# In 4.7.2 the native plugin opens its own EditorFileDialog when no data
	# exists. Its file_selected signal calls the editor's actual C++ baker.
	print("NIGHTGRID_BAKE_PILOT: Native toolbar found; opening save dialog")
	button.pressed.emit()
	await get_tree().process_frame
	var dialogs := _find(button, "EditorFileDialog")
	if dialogs.size() != 1 or not (dialogs[0] as EditorFileDialog).visible:
		_stop("Native bake save dialog not found", 1)
		return
	var dialog := dialogs[0] as EditorFileDialog
	dialog.hide()
	# The native editor can reimport textures after entering 3D. Scanning
	# and importing are separate states in 4.7.2; check both, including
	# after the save dialog opens, before calling the synchronous baker.
	if not await _wait_filesystem_idle():
		_stop("Native import did not settle before bake", 1)
		return
	if not is_instance_valid(dialog) or not is_instance_valid(gi):
		_stop("Bake selection disappeared while waiting for imports", 1)
		return
	print("NIGHTGRID_BAKE_PILOT: Invoking native baker")
	dialog.file_selected.emit(output_path)
	print("NIGHTGRID_BAKE_PILOT: Native baker returned")
	# The native signal is synchronous. Never accept a log message alone.
	await get_tree().process_frame
	var data := gi.light_data
	if data == null or data.get_user_count() != int(inputs.get("receivers", 0)) or data.lightmap_textures.is_empty() or not FileAccess.file_exists(output_path):
		_stop("No fresh bound directional atlas produced", 1)
		return
	if not data.is_using_spherical_harmonics():
		_stop("Expected directional bake", 1)
		return
	if EditorInterface.save_scene() != OK:
		_stop("Could not save baked pilot scene", 1)
		return
	var marker := FileAccess.open("res://bake-result.json", FileAccess.WRITE)
	if marker == null:
		_stop("Could not write result", 1)
		return
	marker.store_string(JSON.stringify({"status":"candidate-only","scene":relative_scene,
		"users":data.get_user_count(),"quality":"high","renderer":RenderingServer.get_current_rendering_method(),
		"adapter":RenderingServer.get_video_adapter_name()}))
	marker.close()
	_stop("Native bake candidate produced; independent validation required", 0)

func _wait_filesystem_idle() -> bool:
	var fs := EditorInterface.get_resource_filesystem()
	var deadline := Time.get_ticks_msec() + 120000
	var quiet_since := -1
	while Time.get_ticks_msec() < deadline:
		await get_tree().process_frame
		if fs.is_scanning() or fs.is_importing():
			quiet_since = -1
			continue
		var now := Time.get_ticks_msec()
		if quiet_since < 0:
			quiet_since = now
		elif now - quiet_since >= 2000:
			print("NIGHTGRID_BAKE_PILOT: Filesystem idle (scan and import)")
			return true
	return false

func _stop(message: String, code: int) -> void:
	print("NIGHTGRID_BAKE_PILOT: ", message)
	get_tree().quit(code)
